 64    def get_angle_over_time(self) -> ArrayLike:
 65        """Get angle over time.
 66
 67        Calculation is dependent on the frequency format (constant scalar number or gradient).
 68        """
 69        if isinstance(self.f, Number):
 70            return 2 * np.pi * self.f * (self.t - self.t[0]) + self.phi
 71        else:
 72            frequency_integrated = (
 73                np.cumsum(2 * np.pi * self.f[1:] * np.diff(self.t)) + self.phi
 74            )
 75            return np.array([self.phi, *frequency_integrated])

114def superimpose_and_concat_periodic_signals(
115    signals: list[list[PeriodicSignal]],
116    num_phases: int = 1,
117    angle_continuation: bool = True,
118    angular_jump: None | ArrayLike = None,
119) -> pd.Series | pd.DataFrame:
120    """Concatenate array of periodic signals.
121
122    Args:
123        signals (list[list[PeriodicSignal]]): List of lists of PeriodicSignal objects. The PeriodicSignals in each row are superimposed and the resulting column is concatenated.
124
125        num_phases (int, optional): To create multiples phases with angle separation of pi/num_phases. Defaults to 1.
126
127        angle_continuation (bool, optional): If true, jumps in phase angle at the transition between signals that are concatenated are avoided (the initial phase angle PeriodicSignal.phi is adapted to be continuous with the previous signal). Defaults to True.
128
129        angular_jump (None | ArrayLike, optional): Intended angular jump at the transition between signals. Length must be the number of rows in 'signals' minus 1. Defaults to None.
130
131    Returns:
132         pd.Series | pd.DataFrame: Series or DataFrame depending on the number of phases.
133
134    Examples:
135        Define array of PeriodicSignals. Rows will be superimposed and the resulting column will be concatenated:
136
137        ```
138        step_size = 1e-3
139        signals = [
140            [
141                PeriodicSignal(np.arange(0, 0.1, step_size), f=(50, 52)),
142                PeriodicSignal(np.arange(0, 0.1, step_size), f=5*50, mag=(0.03, 0.06)),
143            ],
144            [
145                PeriodicSignal(np.arange(0.1, 0.3, step_size), f=50),
146                PeriodicSignal(np.arange(0.1, 0.3, step_size), f=7*50, mag=0.06),
147            ],
148            [
149                PeriodicSignal(np.arange(0.3, 0.5, step_size), f=49),
150                PeriodicSignal(np.arange(0.3, 0.5, step_size), f=9*50, mag=0.06),
151            ],
152        ]
153        res = concat_periodic_signals(signals, num_phases = 3)
154        ```
155    """
156    try:
157        signals = np.array(signals)
158    except ValueError:
159        raise ValueError(
160            "Items in 'signals' must be of same length (same number of Periodic signals in each row)."
161        )
162    if angular_jump is not None:
163        angle_continuation = (
164            True  # Is used to get continuous angle and then add the angular jmp
165        )
166        angular_jump = np.array(angular_jump)
167        angular_jump = np.insert(angular_jump, 0, 0, axis=0)  # first jump is zero
168    elif angle_continuation:
169        angular_jump = [0] * signals.shape[0]
170    phase_signals = [None] * num_phases
171    for phase in range(num_phases):
172        phase_angle = phase * (-2 * np.pi / num_phases)
173        series_to_concat = [None] * signals.shape[0]
174        if angle_continuation:
175            phi_initial_next_row = [
176                signal.phi for signal in signals[0]
177            ]  # is updated for each row
178        for n, signals_to_superpose in enumerate(signals):
179            signals_superposed = [None] * signals.shape[1]
180            for m, signal in enumerate(signals_to_superpose):
181                phi_initial = signal.phi  # store to later reset original value
182                if angle_continuation:
183                    signal.phi = (
184                        phi_initial_next_row[m] + angular_jump[n]
185                    )  # set from last row
186                signal.phi += phase_angle
187                angle = signal.get_angle_over_time()
188                signals_superposed[m] = signal.get_signal_series_from_angle(angle)
189                if (
190                    angle_continuation
191                ):  # store phi_initial_next_row[m] to use as initial angle in next row
192                    phi_initial_next_row[m] = (
193                        angle[-1] + (angle[-1] - angle[-2]) - phase_angle
194                    )  # ? why
195                    # phi_initial_next_row[m] = angle[-1] - phase_angle
196                # ic(angular_jump[n])
197                signal.phi = phi_initial
198            series_to_concat[n] = superpose_series(signals_superposed)
199
200        phase_signals[phase] = pd.concat(series_to_concat, axis=0)
201    if num_phases == 1:
202        return phase_signals[0]
203    else:
204        return pd.concat(phase_signals, axis=1)

253def make_monotonously_increasing(t: ArrayLike, sample_time: float) -> ArrayLike:
254    """Make time 't' monotonously increasing.
255
256    Two consecutive values n, k in 't', where k <= n, are replaced with n, k+sample_time.
257
258    Args:
259        t (ArrayLike): time
260        sample_time (float): sample time
261
262    Returns:
263        ArrayLike: monotonously increasing time
264    """
265    t = np.array(t, dtype=float)
266    diff_t_is_zero = np.concat([[False], np.diff(t) <= 0])
267    t[diff_t_is_zero] = t[diff_t_is_zero] + sample_time
268    return t

271def get_interpolated(
272    t: ArrayLike, y: ArrayLike | list[ArrayLike], sample_time: float
273) -> pd.DataFrame | pd.Series:
274    """Linear interpolated y (specified at 'sample_time') over 't'.
275
276    Args:
277        t (ArrayLike): time
278
279        y (ArrayLike | list[ArrayLike]): y values (length of first dimension matches 'sample_time') or list of y values to get several results.
280
281        sample_time (float): sample time (length of first dimension matches 'y')
282
283    Returns:
284        pd.DataFrame | pd.Series: interpolated values
285    """
286    index = np.arange(t[0], t[-1] + sample_time, sample_time)
287    if not isinstance(y, list):
288        return_series = True
289        y = [y]
290    else:
291        return_series = False
292    ts = pd.DataFrame(np.nan, index=index, columns=range(len(y)))
293    ts.iloc[get_index(ts, t), :] = np.transpose(y)
294    ts.interpolate(inplace=True)
295    ts.index.name = "time"
296    if return_series:
297        ts = ts.squeeze()
298    return ts

302    """Mirror 'ts' at 'y'.

323def discretize(
324    ts: pd.DataFrame | pd.Series, delta_t: float = 1e-6
325) -> pd.DataFrame | pd.Series:
326    """Create a time series that approximates discrete value steps with samples that are close in time.
327
328    For example a 'ts' with index [0,2,3] and a column [20, 50, 60] is changed to a 'ts' with index [0, 1.999, 2, 2.999, 3] and column [20, 20, 50, 50, 60]
329
330    Args:
331        ts: Time series
332        delta_t (float, optional): time difference at discrete steps. Defaults to 1e-6.
333
334    Returns:
335        pd.DataFrame | pd.Series: time series with discrete steps
336    """
337    is_frame = isinstance(ts, pd.DataFrame)
338    ts2 = ts.copy()
339    ts2.index = ts.index - delta_t
340    ts = pd.concat([ts, ts2])
341    ts.sort_index(inplace=True)
342    if is_frame:
343        return pd.DataFrame(
344            ts.iloc[0:-1, :].to_numpy(), index=ts.index[1:], columns=ts.columns
345        )
346    else:
347        return pd.Series(ts.iloc[0:-1].to_numpy(), index=ts.index[1:], name=ts.name)

350def create_ts_from_dict_with_varying_length(
351    d: dict,
352    time_label="time",
353) -> pd.DataFrame:
354    """Create a time series from a dictionary where the values (vectors) can be shorter than the time vector. If the values are shorter than the time vector, they are filled with their last value.
355
356    Args:
357        d (dict): dict
358        time_label (str, optional): Label (key) in dict that indicates the time. Defaults to "time".
359
360    Returns:
361        pd.DataFrame: Time series
362    """
363    df = pd.DataFrame()
364    time = d[time_label]
365    df.index = time
366    df.index.name = time_label
367    del d[time_label]
368    for variable, values in d.items():
369        if len(values) < len(time):
370            diff_len = len(time) - len(values)
371            values = np.concat([values, np.repeat(values[-1], diff_len)])
372        df[variable] = values
373    return df

376def add_normal_noise(
377    df: pd.DataFrame | pd.Series, scale: float, mean: float = 0, seed: None | int = None
378) -> pd.DataFrame | pd.Series:
379    """Add normal distributed noise to time series.
380
381    Thin wrapper around numpy's random number generation.
382
383    Args:
384        df (pd.DataFrame | pd.Series): time series
385        scale (float): Standard deviation (spread or "width") of the distribution. Must be non-negative.
386        mean (float, optional): Mean ("centre") of the distribution.. Defaults to 0.
387        seed (None | int, optional): A seed to initialize the BitGenerator and make results reproducible. If None, then fresh, unpredictable entropy will be pulled from the OS.  Defaults to None.
388
389    Returns:
390        pd.DataFrame: time series with noise
391    """
392    rng = np.random.default_rng(seed=seed)
393    return df + rng.normal(mean, scale, df.shape)
